\subsection{部署前核查清单}

\begin{knowledgebox}{Operations readiness checklist}
\begin{itemize}
    \item 归档数据版本 ID 与 reward log。
    \item Training job config 附带 human sign-off。
    \item Monitoring dashboard 设定 alert threshold。
    \item Governance doc 记录 expected failure modes。
\end{itemize}
\end{knowledgebox}

\section{快捷键与命令速查}

\begin{table}[H]
\centering
\caption{最常用的快捷键}
\begin{tabular}{ll}
\toprule
\textbf{快捷键} & \textbf{功能} \\
\midrule
\texttt{Shift+Tab} & 切换权限模式 / 切换 Plan Mode \\
\texttt{Ctrl+G} & 在编辑器中打开并编辑计划 \\
\texttt{Ctrl+B} & 将子 Agent 放到后台运行 \\
\texttt{Ctrl+V} & 粘贴图片到对话 \\
\texttt{Option+K / Alt+K} & (VS Code) 快速插入 @文件引用 \\
\texttt{Cmd+Shift+Esc} & (VS Code) 打开新对话标签 \\
\bottomrule
\end{tabular}
\end{table}

\begin{table}[H]
\centering
\caption{最常用的斜杠命令}
\begin{tabular}{ll}
\toprule
\textbf{命令} & \textbf{功能} \\
\midrule
\texttt{/clear} & 清空上下文，保留 CLAUDE.md \\
\texttt{/compact} & 压缩上下文，可加焦点参数 \\
\texttt{/context} & 查看上下文使用情况 \\
\texttt{/memory} & 查看和管理 Memory \\
\texttt{/mcp} & 查看 MCP server 及其开销 \\
\texttt{/init} & 自动分析代码库生成 CLAUDE.md 初稿 \\
\texttt{/rewind} & 回退到任意 checkpoint \\
\texttt{/freeze} & 限制编辑范围到指定目录 \\
\bottomrule
\end{tabular}
\end{table}

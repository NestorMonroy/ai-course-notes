\section{总结与延伸}

\subsection{三条桥接路径}

\begin{enumerate}
    \item \textbf{非形式化思考 + 形式化证明}（LeanSTaR）：训练模型在证明步骤前进行非形式化推理
    \item \textbf{非形式化证明 + 形式化填补}（DSP + LeanHammer）：先生成高级草图，再用自动化工具填补细节
    \item \textbf{研究级辅助}（MiniCTX + LLMLean）：面向真实数学项目的上下文感知证明和实用工具
\end{enumerate}

\subsection{拓展阅读}
\begin{itemize}
    \item LeanSTaR（ICLR 2025）：RL 训练模型生成非形式化思考
    \item Draft, Sketch, Prove（ICLR 2023）：草图-填补证明方法
    \item MiniCTX（ICLR 2025 Oral）：上下文依赖定理证明基准
    \item LLMLean：\url{https://github.com/cmu-l3/llmlean}
    \item DeepSeek-Prover：融合非形式化思考的 SOTA 证明器
\end{itemize}
